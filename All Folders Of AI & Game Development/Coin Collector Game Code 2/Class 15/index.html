The Game
10 enemy characters are standing on the map. Each one has a ClickDetector — aim and click to shoot. Normal enemies give 10 points, gold enemies give 50 points, and boss enemies take 3 hits to defeat. Shoot all 10 to win. Your score updates live on the leaderboard.

How to Build — Step by Step
  • 1
    Open Studio → Baseplate → ServerScriptService → Script
    All the code goes in one script. No Parts need to be placed manually — the script builds everything.
  • 2
    The script creates the leaderboard first
    A Score stat and a Kills stat are created for every player who joins.
  • 3
    A function builds one enemy character
    The createEnemy() function makes a character from Parts — a head, body, arms, and legs. It takes position, color, health, and points as parameters.
  • 4
    A for loop calls createEnemy() for all 10 enemies
    7 normal, 2 gold, 1 boss — each placed in a different spot across the map.
  • 5
    Press F5 — aim and click the enemies to shoot them
    Click the head or body of each enemy. Watch the score go up on the leaderboard. The boss needs 3 clicks — it shows its remaining HP each time.

Part 1 — Leaderboard + shoot sound leaderstats
local Players = game:GetService("Players")

-- Shoot sound
local shootSound = Instance.new("Sound")
shootSound.Parent = workspace
shootSound.SoundId = "rbxassetid://5853106423"
shootSound.Volume = 1

-- Win sound
local winSound = Instance.new("Sound")
winSound.Parent = workspace
winSound.SoundId = "rbxassetid://4612355792"
winSound.Volume = 1

local totalEnemies = 10
local defeated = 0
local currentPlayer = nil

Players.PlayerAdded:Connect(function(player)
  currentPlayer = player
  local ls = Instance.new("Folder")
  ls.Name = "leaderstats"
  ls.Parent = player
  local score = Instance.new("IntValue")
  score.Name = "Score" ; score.Value = 0
  score.Parent = ls
  local kills = Instance.new("IntValue")
  kills.Name = "Kills" ; kills.Value = 0
  kills.Parent = ls
end)
Part 2 — createEnemy() function builds character
local function createEnemy(pos, color, hp, pts)

  -- MODEL to hold all body parts
  local model = Instance.new("Model")
  model.Name = "Enemy"
  model.Parent = workspace

  local bc = BrickColor.new(color)

  -- HEAD
  local head = Instance.new("Part")
  head.Size = Vector3.new(2,2,2)
  head.BrickColor = bc
  head.Shape = Enum.PartType.Ball
  head.Material = Enum.Material.Neon
  head.Anchored = true
  head.Position = Vector3.new(pos.X, pos.Y + 4, pos.Z)
  head.Parent = model

  -- BODY
  local body = Instance.new("Part")
  body.Size = Vector3.new(2,3,1)
  body.BrickColor = bc
  body.Anchored = true
  body.Position = Vector3.new(pos.X, pos.Y + 1.5, pos.Z)
  body.Parent = model

  -- LEFT LEG
  local legL = Instance.new("Part")
  legL.Size = Vector3.new(0.8,2,0.8)
  legL.BrickColor = bc
  legL.Anchored = true
  legL.Position = Vector3.new(pos.X - 0.6, pos.Y - 1, pos.Z)
  legL.Parent = model

  -- RIGHT LEG
  local legR = Instance.new("Part")
  legR.Size = Vector3.new(0.8,2,0.8)
  legR.BrickColor = bc
  legR.Anchored = true
  legR.Position = Vector3.new(pos.X + 0.6, pos.Y - 1, pos.Z)
  legR.Parent = model

  -- CLICKDETECTOR on the body
  local detector = Instance.new("ClickDetector")
  detector.MaxActivationDistance = 100
  detector.Parent = body

  local currentHp = hp

  detector.MouseClick:Connect(function(player)

    currentHp = currentHp - 1
    shootSound:Play()

    if currentHp > 0 then
      -- boss still alive — flash white
      body.BrickColor = BrickColor.new("White")
      task.wait(0.1)
      body.BrickColor = bc
      print("💥 Boss HP left: " .. currentHp)
    else
      -- enemy defeated!
      player.leaderstats.Score.Value += pts
      player.leaderstats.Kills.Value += 1
      defeated += 1
      print("💀 +" .. pts .. " pts! Score: " .. player.leaderstats.Score.Value)
      model:Destroy()

      if defeated == totalEnemies then
        winSound:Play()
        print("══════════════════════")
        print("🏆 " .. player.Name .. " WINS!")
        print(" Final Score: " .. player.leaderstats.Score.Value)
        print("══════════════════════")
      end
    end
  end)
end
Part 3 — Spawn all 10 enemies for loop
-- Spawn 7 Normal enemies
for i = 1, 7 do
  local pos = Vector3.new(i * 8, 4, 10)
  createEnemy(pos, "Really red", 1, 10)
end

-- Spawn 2 Gold enemies
for i = 1, 2 do
  local pos = Vector3.new(i * 20, 4, -10)
  createEnemy(pos, "Bright yellow", 1, 50)
end

-- Spawn 1 Boss — 3 HP, 100 pts
local bossPos = Vector3.new(30, 4, 0)
createEnemy(bossPos, "Royal purple", 3, 100)

print("🎯 10 enemies spawned!")
print(" Click to shoot them!")
