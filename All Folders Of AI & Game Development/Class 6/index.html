The Dungeon Escape
The player is trapped in a dungeon. They must pass through 4 rooms — each room has a challenge. Each room checks the player's stats using if/else and decides what happens. If the player survives all 4 rooms with health above 0, they escape and win. Every single decision in this game is made by a condition.

BLOCK 1 — Variables + Part References Setup
-- ══════════════════════════════

-- PLAYER STATS
local playerName = "Ahmed"
local health = 100
local maxHealth = 100
local attackPower = 40
local armour = 10
local gold = 80
local score = 0
local hasKey = false
local isAlive = true

-- PART REFERENCES
local entryGate = workspace.EntryGate
local fireRoom = workspace.FireRoom
local lockedDoor = workspace.LockedDoor
local bossRoom = workspace.BossRoom
local rankRoom = workspace.RankRoom
local escapePortal= workspace.EscapePortal
BLOCK 2 — Welcome Screen print()
print("════════════════════════════")
print(" 🏰 THE DUNGEON ESCAPE")
print("════════════════════════════")
print("Player : " .. playerName)
print("Health : " .. health .. " / " .. maxHealth)
print("Attack : " .. attackPower)
print("Armour : " .. armour)
print("Gold : " .. gold)
print("Key : " .. tostring(hasKey))
print("════════════════════════════")
BLOCK 3 — Entry Gate if / else
-- 🚪 ENTRY GATE
-- Need at least 50 HP to enter
print("🚪 Approaching the gate...")

if health >= 50 then
  entryGate.BrickColor = BrickColor.new("Lime green")
  entryGate.Transparency = 0.5
  print("✅ Gate OPEN — Enter, " .. playerName)
  score = score + 10
else
  entryGate.BrickColor = BrickColor.new("Bright red")
  print("❌ Gate LOCKED — Too weak!")
  isAlive = false
end
BLOCK 4 — Fire Room if / elseif / else
-- 🔥 FIRE ROOM
-- Armour reduces fire damage
local fireDamage = 50 - armour -- armour reduces it

fireRoom.BrickColor = BrickColor.new("Dark orange")
fireRoom.Material = Enum.Material.Neon

print("🔥 Fire Room! Damage: " .. fireDamage)

if armour >= 30 then
  print("🛡️ Heavy armour — no damage!")
  score = score + 20
elseif armour >= 10 then
  health = health - fireDamage
  print("🟡 Partial armour — took " .. fireDamage .. " dmg")
  print(" Health: " .. health)
else
  health = health - 50
  print("🔴 No armour! Full 50 damage!")
  print(" Health: " .. health)
end

if health <= 0 then
  isAlive = false
  print("💀 Burned alive!")
end
BLOCK 5 — Locked Door if + or
-- 🔐 LOCKED DOOR
-- Need key OR 100+ gold to pass
local doorCost = 100

if hasKey == true or gold >= doorCost then
  lockedDoor.BrickColor = BrickColor.new("Lime green")
  lockedDoor.Transparency = 0.8
  print("🔓 Door OPEN!")
  if hasKey == true then
    print(" Used the key!")
  else
    gold = gold - doorCost
    print(" Paid gold. Left: " .. gold)
  end
  score = score + 20
else
  lockedDoor.BrickColor = BrickColor.new("Dark red")
  print("🔒 LOCKED! Need key or 100 gold")
  print(" You have: " .. gold .. " gold")
  health = health - 20 -- penalty!
  print(" -20 HP penalty! Health: " .. health)
end
BLOCK 6 — Boss Fight if + and
-- 🧌 BOSS FIGHT
local bossHealth = 35
local bossAttack = 25

bossRoom.BrickColor = BrickColor.new("Hot pink")
bossRoom.Material = Enum.Material.Neon
print("🧌 BOSS appears! HP: " .. bossHealth)

if attackPower >= bossHealth then
  bossRoom.BrickColor = BrickColor.new("Lime green")
  print("⚔️ Boss DEFEATED in one hit!")
  score = score + 30
elseif attackPower >= bossHealth / 2 then
  health = health - bossAttack
  print("🤜 Boss wounded — survived but took " .. bossAttack .. " dmg")
  print(" Health: " .. health)
  score = score + 15
else
  health = health - (bossAttack * 2)
  print("💀 Too weak! Boss dealt double damage!")
  print(" Health: " .. health)
end

if health <= 0 then
  isAlive = false
end
BLOCK 7 — Score Rank Room elseif chain
-- 🏆 RANK ROOM
print("🏆 Final Score: " .. score)

if score >= 70 then
  rankRoom.BrickColor = BrickColor.new("Bright yellow")
  rankRoom.Material = Enum.Material.Neon
  print("🥇 Rank: GOLD — Legendary!")
elseif score >= 40 then
  rankRoom.BrickColor = BrickColor.new("White")
  print("🥈 Rank: SILVER — Well done!")
elseif score >= 20 then
  rankRoom.BrickColor = BrickColor.new("Dark orange")
  print("🥉 Rank: BRONZE — Keep going!")
else
  rankRoom.BrickColor = BrickColor.new("Medium stone grey")
  print("❌ No rank — Try again!")
end
BLOCK 8 — Win / Game Over Screen (final block — goes at the very end) if / else — FINAL
-- 🌟 WIN / GAME OVER CHECK
print("════════════════════════════")

if isAlive == true and health > 0 then
  escapePortal.BrickColor = BrickColor.new("Cyan")
  escapePortal.Material = Enum.Material.Neon
  escapePortal.Transparency = 0.3
  print("🌟 YOU ESCAPED THE DUNGEON!")
  print(" Player : " .. playerName)
  print(" Health : " .. health .. " / " .. maxHealth)
  print(" Score : " .. score)
  print(" Gold : " .. gold)
else
  escapePortal.BrickColor = BrickColor.new("Black")
  escapePortal.Transparency = 0.8
  print("💀 GAME OVER!")
  print(" " .. playerName .. " did not survive.")
  print(" Final health: " .. health)
  print(" Try again!")
end
print("════════════════════════════")

Entry Gate

health ≥ 50 → open (green, transparent)
else → locked (red), isAlive = false
🔥
Fire Room
elseif chain elseif
armour ≥ 30 → no damage
armour ≥ 10 → partial damage
else → full 50 damage
🔐
Locked Door
if + or or
hasKey OR gold ≥ 100 → open
else → -20 HP penalty, stays red
🧌
Boss Fight
elseif elseif
attack ≥ bossHP → one-shot win
attack ≥ half → survive with damage
else → double damage
🏆
Rank Room
elseif chain elseif
score ≥ 70 → Gold (neon yellow)
≥ 40 → Silver (white)
≥ 20 → Bronze · else → no rank
🌟
Escape Portal
if + and and
isAlive AND health > 0 → win (cyan neon portal)
else → game over (black + transparent)
💀
Death Check
if after damage if
After every damage event: if health ≤ 0 then isAlive = false. Stops the game immediately.
